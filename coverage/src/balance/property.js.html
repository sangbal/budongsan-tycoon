/**
 * Seoul Survival - 부동산 밸런스 설정
 *
 * 이 파일에서 부동산의 기본 가격과 월세(초당) 수익을 조정할 수 있습니다.
 *
 * 가격 공식: 실제 가격 = baseCost * (1.05 ^ 보유개수)
 * - 예: 빌라 2번째 구매 시 = 2.5억 * 1.05 = 2.625억
 * - 서울타워는 예외: 고정 가격 (프레스티지 아이템)
 */
 
// 부동산 기본 가격 (첫 구매 가격)
// main.js에서 BASE_COSTS로 사용됨
export const BASE_COSTS = {
  villa: 250_000_000, // 빌라: 2.5억원
  officetel: 350_000_000, // 오피스텔: 3.5억원
  apartment: 800_000_000, // 아파트: 8억원
  shop: 1_200_000_000, // 상가: 12억원
  building: 3_000_000_000, // 빌딩: 30억원
  tower: 1_000_000_000_000, // 서울타워: 1조원 (프레스티지)
}
 
// 부동산 초당 월세 수익 기본값
// main.js에서 이 값으로 BASE_RENT를 초기화함 (업그레이드로 변경 가능)
export const DEFAULT_BASE_RENT = {
  villa: 84_380, // 빌라: 84,380원/초
  officetel: 177_190, // 오피스텔: 177,190원/초
  apartment: 607_500, // 아파트: 607,500원/초
  shop: 1_370_000, // 상가: 1,370,000원/초
  building: 5_140_000, // 빌딩: 5,140,000원/초
  // 서울타워는 수익 없음 (프레스티지 전용)
}
 
// 가격 성장률 (구매할 때마다 가격이 이 비율로 증가)
export const PROPERTY_COST_GROWTH = 1.05 // 5% 증가
 
// 판매 환급률 (100% = 구매가 전액 환불)
export const PROPERTY_SELL_RATE = 1.0 // 100% 환급
 
// 서울타워 특수 설정
export const TOWER_CONFIG = {
  price: 1_000_000_000_000, // 1조원
  hasIncome: false, // 수익 없음
  isPrestige: true, // 프레스티지 아이템
  sellable: false, // 판매 불가
}
 
// ===== 수익률 분석 (참고용) =====
// 상품        | 기본가격   | 초당수익     | ROI(초)  | ROI(분)
// 빌라        | 2.5억     | 84,380      | 2,963초  | 49.4분
// 오피스텔    | 3.5억     | 177,190     | 1,975초  | 32.9분
// 아파트      | 8억       | 607,500     | 1,317초  | 21.9분
// 상가        | 12억      | 1,370,000   | 876초    | 14.6분
// 빌딩        | 30억      | 5,140,000   | 584초    | 9.7분
 
// 해금 순서: 빌라 → 오피스텔 → 아파트 → 상가 → 빌딩 → 서울타워
