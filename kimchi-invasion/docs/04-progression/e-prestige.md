### 4.8.4. 랜덤 행성 생성 시스템

```javascript
// 행성 생성 알고리즘 (시드 기반)
function generatePlanet(seed) {
  const rng = new SeededRandom(seed)

  return {
    name: generatePlanetName(rng),

    // 자원 분포 (0.0 ~ 1.0, 0.5가 평균)
    resources: {
      iron_density: rng.range(0.2, 1.0), // 철광석 밀도
      ice_distribution: rng.range(0.1, 0.9), // 얼음 분포
      fertility: rng.range(0.1, 1.0), // 토양 비옥도
      rare_elements: rng.range(0.0, 0.8), // 희귀 자원
    },

    // 환경 요소
    environment: {
      terrain_roughness: rng.range(0.0, 1.0), // 지형 험준도
      dust_storm_frequency: rng.range(0.1, 0.5), // 폭풍 빈도
      solar_efficiency: rng.range(0.6, 1.2), // 태양광 효율
      radiation_level: rng.range(0.3, 1.5), // 방사선 수준
    },

    // 시작 자원 (비옥도 기반)
    starting_resources: {
      iron_plate: Math.floor(50 + rng.range(0, 50) * iron_density),
      water: Math.floor(30 + rng.range(0, 30) * ice_distribution),
    },

    // 특수 이벤트 확률 조정
    event_modifiers: {
      meteor_shower: rng.range(0.5, 1.5),
      solar_flare: rng.range(0.5, 1.5),
      dust_devil: rng.range(0.5, 1.5),
    },
  }
}

// 행성 난이도 계산
function calculateDifficulty(planet) {
  const resourceScore =
    planet.resources.iron_density * 0.3 +
    planet.resources.ice_distribution * 0.2 +
    planet.resources.fertility * 0.3 +
    planet.resources.rare_elements * 0.2

  const environmentPenalty =
    planet.environment.terrain_roughness * 0.3 +
    planet.environment.dust_storm_frequency * 0.3 +
    (1 - planet.environment.solar_efficiency) * 0.2 +
    planet.environment.radiation_level * 0.2

  const difficulty = (1 - resourceScore + environmentPenalty) / 2

  // 난이도 등급
  if (difficulty < 0.3) return 'Easy'
  if (difficulty < 0.5) return 'Normal'
  if (difficulty < 0.7) return 'Hard'
  return 'Extreme'
}
```

### 4.8.6. 프레스티지 영구 보너스

```javascript
// 프레스티지 보너스 계산
const PRESTIGE_BONUSES = {
  // 기본 보너스 (매 프레스티지마다)
  per_prestige: {
    production_speed: 0.05, // 생산 속도 +5%/회
    research_speed: 0.03, // 연구 속도 +3%/회
    starting_money: 500, // 시작 자금 +$500/회
  },

  // 누적 상한
  caps: {
    production_speed: 1.0, // 최대 +100% (20회)
    research_speed: 0.6, // 최대 +60% (20회)
    starting_money: 10000, // 최대 +$10,000 (20회)
  },

  // 마일스톤 보너스
  milestones: {
    5: { type: 'unlock', value: 'blueprint_mode' }, // 5회: 블루프린트
    10: { type: 'starting_money', value: 1000 }, // 10회: 시작자금+$1000
    15: { type: 'unlock', value: 'auto_research' }, // 15회: 자동 연구
    20: { type: 'loadout_slot', value: 1 }, // 20회: 슬롯+1
    25: { type: 'unlock', value: 'time_acceleration' }, // 25회: 시간 가속
    30: { type: 'production_speed', value: 0.25 }, // 30회: 생산+25% 추가
  },
}

// 총 보너스 계산
function calculateTotalBonus(prestigeCount) {
  const bonus = {
    production_speed: Math.min(
      prestigeCount * PRESTIGE_BONUSES.per_prestige.production_speed,
      PRESTIGE_BONUSES.caps.production_speed
    ),
    research_speed: Math.min(
      prestigeCount * PRESTIGE_BONUSES.per_prestige.research_speed,
      PRESTIGE_BONUSES.caps.research_speed
    ),
    starting_money: Math.min(
      prestigeCount * PRESTIGE_BONUSES.per_prestige.starting_money,
      PRESTIGE_BONUSES.caps.starting_money
    ),
    unlocks: [],
    extra_loadout_slots: 0,
  }

  // 마일스톤 보너스 적용
  for (const [count, milestone] of Object.entries(PRESTIGE_BONUSES.milestones)) {
    if (prestigeCount >= parseInt(count)) {
      if (milestone.type === 'unlock') {
        bonus.unlocks.push(milestone.value)
      } else if (milestone.type === 'loadout_slot') {
        bonus.extra_loadout_slots += milestone.value
      } else if (milestone.type === 'production_speed') {
        bonus.production_speed += milestone.value
      } else if (milestone.type === 'starting_money') {
        bonus.starting_money += milestone.value
      }
    }
  }

  return bonus
}
```
