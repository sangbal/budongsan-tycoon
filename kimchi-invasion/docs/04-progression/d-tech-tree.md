## 4.7. 기술 트리 (Technology Research)

### 4.7.1. 연구 시스템 메커니즘

```javascript
// 연구 시스템 파라미터
const RESEARCH_SYSTEM = {
  // 연구소 처리 속도
  lab_speed: {
    basic_lab: 1.0, // 기본 연구소: 1x
    advanced_lab: 2.0, // 고급 연구소: 2x
    quantum_lab: 5.0, // 양자 연구소: 5x (Tier 5)
  },

  // 연구 시간 계산
  // 연구_시간 = 기본_시간 × (1 / 연구소_속도_합) × (1 - 보너스)
  base_time_per_point: 30, // 연구점 1개당 30초

  // 연구 큐
  queue: {
    max_queue: 5, // 최대 5개 연구 대기
    auto_start: true, // 완료 시 다음 자동 시작
  },

  // 동시 연구
  parallel_research: {
    tier1_2: false, // Tier 1-2: 순차 연구만
    tier3_4: true, // Tier 3-4: 2개 동시 연구
    tier5: true, // Tier 5: 3개 동시 연구
  },
}

// 예시: 컨베이어 기술 (🔬×15)
// 기본 연구소 1개: 15 × 30초 = 450초 (7.5분)
// 기본 연구소 2개: 450 / 2 = 225초 (3.75분)
// 고급 연구소 1개: 450 / 2 = 225초 (3.75분)
```
