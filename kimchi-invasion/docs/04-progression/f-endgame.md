### 4.9.4. 무한 프레스티지 시스템

```javascript
// 무한 프레스티지 보너스 (30회 이후)
const INFINITE_PRESTIGE = {
  // 30회까지는 기존 보너스 (4.8.6 참조)
  cap_at: 30,

  // 30회 이후 무한 스케일링
  after_cap: {
    // 매 회 추가 보너스 (상한 없음)
    production_speed: 0.01, // +1%/회 (무한)
    research_speed: 0.005, // +0.5%/회 (무한)

    // 특별 마일스톤 (계속 추가)
    milestones: {
      50: { type: 'unlock', value: 'time_warp_x2' }, // 시간 2배속
      75: { type: 'unlock', value: 'auto_expand' }, // 자동 확장
      100: { type: 'title', value: '김치 황제' }, // 전설 칭호
      150: { type: 'unlock', value: 'time_warp_x5' }, // 시간 5배속
      200: { type: 'title', value: '화성의 주인' }, // 최상위 칭호
      500: { type: 'unlock', value: 'creative_mode' }, // 크리에이티브 모드
    },
  },

  // 리더보드 순위 계산
  // 순위 점수 = 프레스티지 횟수 × 1000 + 최고 SPM
  leaderboard_score: (prestige, maxSPM) => prestige * 1000 + maxSPM,
}

// 예시: 100회 프레스티지 플레이어
// 생산 속도: +100% (30회 상한) + 70% (30~100회) = +170%
// 연구 속도: +60% (30회 상한) + 35% (30~100회) = +95%
// 해금: 블루프린트, 자동연구, 시간가속, 시간2배, 자동확장, 김치황제 칭호
```
