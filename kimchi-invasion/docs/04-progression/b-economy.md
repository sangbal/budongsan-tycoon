### 4.5.3. 가격 계산 공식

```javascript
// 수출 가격 계산
function calculateExportPrice(product, quantity, quality) {
  const BASE_PRICES = {
    kimchi_basic: 10,
    kkakdugi: 12,
    chonggak: 15,
    gat: 18,
    baek: 20,
    mukeunji: 35,
  }

  const QUALITY_MULTIPLIERS = {
    normal: 1.0,
    fine: 1.5,
    premium: 2.5,
    martian_special: 10.0,
  }

  // 기본 가격
  let price = BASE_PRICES[product]

  // 품질 적용
  price *= QUALITY_MULTIPLIERS[quality]

  // 시장 변동 (Sol마다 변경)
  const marketFluctuation = getMarketFluctuation(product) // -0.3 ~ +0.3
  price *= 1 + marketFluctuation

  // 대량 판매 할인
  if (quantity > 100) {
    const saturationPenalty = Math.min(0.5, (quantity - 100) * 0.0005)
    price *= 1 - saturationPenalty
  }

  // 트렌드 보너스 (활성화 시)
  if (isTrending(product)) {
    price *= 1.5
  }

  return Math.floor(price * quantity)
}

// 예시: 명품 묵은지 200캔 (트렌드 중)
// = $35 × 2.5 × 1.1 (시장 +10%) × 0.95 (대량 -5%) × 1.5 (트렌드)
// = $136.97/캔 × 200 = $27,394
```
